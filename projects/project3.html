import pandas as pd

data = [
    {"用户ID": "U001", "行为": "浏览", "页面": "首页"},
    {"用户ID": "U001", "行为": "浏览", "页面": "商品列表"},
    {"用户ID": "U001", "行为": "浏览", "页面": "商品详情"},
    {"用户ID": "U001", "行为": "加购", "页面": "商品详情"},
    {"用户ID": "U001", "行为": "浏览", "页面": "购物车"},
    {"用户ID": "U001", "行为": "下单", "页面": "结算页"},
    {"用户ID": "U002", "行为": "浏览", "页面": "首页"},
    {"用户ID": "U002", "行为": "浏览", "页面": "商品列表"},
    {"用户ID": "U002", "行为": "浏览", "页面": "商品详情"},
    {"用户ID": "U003", "行为": "浏览", "页面": "首页"},
    {"用户ID": "U003", "行为": "浏览", "页面": "商品列表"},
    {"用户ID": "U003", "行为": "浏览", "页面": "商品详情"},
    {"用户ID": "U003", "行为": "加购", "页面": "商品详情"},
    {"用户ID": "U004", "行为": "浏览", "页面": "首页"},
    {"用户ID": "U004", "行为": "浏览", "页面": "商品列表"},
]

df = pd.DataFrame(data)

# 统计各行为类型的数量
behavior_counts = df['行为'].value_counts()
print("【各行为类型数量】")
print(behavior_counts)

# 计算转化漏斗
total_users = df['用户ID'].nunique()
browse_users = df[df['行为'] == '浏览']['用户ID'].nunique()
cart_users = df[df['行为'] == '加购']['用户ID'].nunique()
order_users = df[df['行为'] == '下单']['用户ID'].nunique()

print(f"\n【转化漏斗】")
print(f"浏览用户数: {browse_users}")
print(f"加购用户数: {cart_users}")
print(f"下单用户数: {order_users}")
print(f"\n浏览→加购转化率: {cart_users/browse_users*100:.1f}%")
print(f"加购→下单转化率: {order_users/cart_users*100:.1f}%")
