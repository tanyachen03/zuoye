import pandas as pd

# 用户购买记录
data = [
    {"order_id": "O001", "user_id": "U001", "user_name": "张三", "date": "2024-01-15", "category": "电子产品", "amount": 2999},
    {"order_id": "O002", "user_id": "U002", "user_name": "李四", "date": "2024-01-16", "category": "服装", "amount": 399},
    {"order_id": "O003", "user_id": "U001", "user_name": "张三", "date": "2024-02-20", "category": "服装", "amount": 599},
    {"order_id": "O004", "user_id": "U003", "user_name": "王五", "date": "2024-02-25", "category": "家居", "amount": 1299},
    {"order_id": "O005", "user_id": "U001", "user_name": "张三", "date": "2024-03-10", "category": "电子产品", "amount": 4599},
    {"order_id": "O006", "user_id": "U002", "user_name": "李四", "date": "2024-03-15", "category": "服装", "amount": 299},
    {"order_id": "O007", "user_id": "U004", "user_name": "赵六", "date": "2024-03-20", "category": "电子产品", "amount": 1999},
]

df = pd.DataFrame(data)

print("=" * 60)
print("【第1步】统计每个用户的购买次数")
user_counts = df.groupby('user_id').size().reset_index(name='purchase_count')
user_counts = user_counts.merge(df[['user_id', 'user_name']].drop_duplicates(), on='user_id')
print(user_counts.to_string(index=False))

print("\n" + "=" * 60)
print("【第2步】购买次数分布")
purchase_dist = user_counts['purchase_count'].value_counts().sort_index()
for count, num in purchase_dist.items():
    print(f"购买 {count} 次的用户: {num} 人")
print(f"\n总用户数: {len(user_counts)}")

print("\n" + "=" * 60)
print("【第3步】计算复购率")
repeat_users = len(user_counts[user_counts['purchase_count'] >= 2])
total_users = len(user_counts)
repeat_rate = repeat_users / total_users * 100
print(f"复购用户数（购买2次及以上）: {repeat_users}")
print(f"整体复购率: {repeat_rate:.1f}%")

print("\n" + "=" * 60)
print("【第4步】Top购买次数用户")
top_users = user_counts.nlargest(5, 'purchase_count')
print(top_users.to_string(index=False))

print("\n" + "=" * 60)
print("【第5步】复购用户消费分析")
repeat_user_ids = user_counts[user_counts['purchase_count'] >= 2]['user_id']
repeat_df = df[df['user_id'].isin(repeat_user_ids)]
repeat_spending = repeat_df.groupby('user_id')['amount'].sum()
print(f"复购用户总消费: {repeat_spending.sum():.0f} 元")
print(f"复购用户平均消费: {repeat_spending.mean():.2f} 元")

分析技巧

value_counts() - 统计各值的出现次数
size() - 统计分组后的记录数
nlargest(n, col) - 返回最大的n行
isin() - 判断是否在列表中
merge() - 数据表合并
