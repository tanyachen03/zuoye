项目背景

广告投放是企业获取客户的重要手段，但不同渠道的效果和成本差异很大。ROI（投资回报率）是衡量广告效果的核心指标，帮助企业优化广告预算分配。

广告投放数据

渠道投放成本(元)曝光量点击量销售额(元)

学习内容

ROI核心公式
ROI = (收入 - 成本) / 成本 × 100%
ROI > 0% 表示盈利，ROI < 0% 表示亏损

抖音

小红书

小贴士
ROI不是唯一指标！应先看CPM(千次曝光成本)再看ROI，区分品牌广告和效果广告。品牌广告重曝光，效果广告重复购。
常见错误
只看短期ROI，忽略了品牌广告的长期心智价值！小红书种草、公众号内容等渠道难以直接归因，但长期价值巨大。

代码示例2：绘制雷达图对比各渠道

import matplotlib.pyplot as plt
import numpy as np

# 计算各维度指标（标准化到0-100）
df['ROI_norm'] = (df['ROI'] - df['ROI'].min()) / (df['ROI'].max() - df['ROI'].min()) * 100
df['CTR'] = (df['clicks'] / df['impressions'] * 100).round(2)
df['CVR'] = (df['orders'] / df['clicks'] * 100).round(2)

# 雷达图
categories = ['ROI', '曝光量', '点击率', '转化率']
angles = np.linspace(0, 2*np.pi, len(categories), endpoint=False).tolist()
angles += angles[:1]

fig, ax = plt.subplots(figsize=(8, 8), subplot_kw=dict(polar=True))
for idx, row in df.iterrows():
    values = [row['ROI_norm'], row['impressions']/10000, row['CTR']*10, row['CVR']*10]
    values += values[:1]
    ax.plot(angles, values, 'o-', linewidth=2, label=row['channel'])
    ax.fill(angles, values, alpha=0.25)

ax.set_xticks(angles[:-1])
ax.set_xticklabels(categories)
plt.legend(loc='upper right')
plt.show()

import pandas as pd

data = [
    {"channel": "抖音", "cost": 50000, "impressions": 500000, "clicks": 25000, "orders": 500, "sales": 150000},
    {"channel": "小红书", "cost": 20000, "impressions": 200000, "clicks": 8000, "orders": 160, "sales": 48000},
    {"channel": "百度SEM", "cost": 40000, "impressions": 400000, "clicks": 20000, "orders": 320, "sales": 96000},
    {"channel": "微信公众号", "cost": 15000, "impressions": 150000, "clicks": 4500, "orders": 90, "sales": 27000},
    {"channel": "B站", "cost": 25000, "impressions": 250000, "clicks": 10000, "orders": 200, "sales": 60000},
]

df = pd.DataFrame(data)

# 第1步：计算各渠道ROI
print("=" * 70)
print("【第1步】计算各渠道ROI")
df['profit'] = df['sales'] - df['cost']
df['ROI'] = ((df['sales'] - df['cost']) / df['cost'] * 100).round(2)
df['CPM'] = (df['cost'] / df['impressions'] * 1000).round(2)
df['CTR'] = (df['clicks'] / df['impressions'] * 100).round(2)
df['CVR'] = (df['orders'] / df['clicks'] * 100).round(2)

print("\n基础指标：")
print(df[['channel', 'cost', 'sales', 'profit', 'ROI']].to_string(index=False))

print(f"\n  CPM(千次曝光成本): {df['CPM'].mean():.2f} 元")
print(f"  平均CTR(点击率): {df['CTR'].mean():.2f}%")
print(f"  平均CVR(转化率): {df['CVR'].mean():.2f}%")

# 第2步：按ROI降序排列
print("\n" + "=" * 70)
print("【第2步】ROI排名")
df_sorted = df.sort_values('ROI', ascending=False).reset_index(drop=True)
print("\n渠道ROI排名：")
print("-" * 50)
max_roi = df_sorted['ROI'].max()
for idx, row in df_sorted.iterrows():
    bar_len = int(row['ROI'] / max_roi * 30) if row['ROI'] > 0 else 0
    bar = "█" * bar_len
    status = "✅" if row['ROI'] > 0 else "❌"
    print(f"{idx+1}. {row['channel']:<8} {bar} {row['ROI']:>6.1f}% {status}")
print("-" * 50)

# 第3步：最优渠道分析
print("\n" + "=" * 70)
print("【第3步】最优渠道分析")
best_channel = df_sorted.iloc[0]
worst_channel = df_sorted.iloc[-1]

print(f"\n🏆 ROI最高渠道: {best_channel['channel']}")
print(f"   - ROI: {best_channel['ROI']:.1f}%")
print(f"   - 投放成本: {best_channel['cost']:,}元")
print(f"   - 带来销售额: {best_channel['sales']:,}元")
print(f"   - 利润: {best_channel['profit']:,}元")

print(f"\n⚠️ ROI最低渠道: {worst_channel['channel']}")
print(f"   - ROI: {worst_channel['ROI']:.1f}%")

# 边际ROI分析
print("\n" + "=" * 70)
print("【第4步】边际ROI分析（每投入1000元的销售额产出）")
df['marginal_roi'] = (df['sales'] / df['cost'] * 1000).round(2)
df_marginal = df.sort_values('marginal_roi', ascending=False)
for _, row in df_marginal.iterrows():
    bar_len = int(row['marginal_roi'] / df_marginal['marginal_roi'].max() * 30)
    bar = "▓" * bar_len
    print(f"   {row['channel']:<8} {bar} {row['marginal_roi']:>6.2f}元")

print("\n" + "=" * 70)
print("【结论与建议】")
print(f"1. 推荐{'增加' if best_channel['ROI'] > 0 else '减少'} {best_channel['channel']} 的预算投放")
print(f"2. {'关注' if worst_channel['ROI'] < 50 else '可适当减少'} {worst_channel['channel']} 的投放")
print(f"3. 整体平均ROI: {df['ROI'].mean():.1f}%")

核心指标

ROI = (收益-成本)/成本×100%
CPM = 成本/曝光量×1000
CTR = 点击量/曝光量×100%
CVR = 订单数/点击量×100%
边际ROI = 销售额/成本×1000
