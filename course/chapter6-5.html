一、案例背景与需求分析

案例背景

某电商平台需要制作一份2024年上半年销售数据分析报告，用于向管理层汇报业务状况。报告需要包含销售趋势、区域分析、产品结构、用户画像等多个维度的可视化分析。

时间范围：2024年1月-6月
数据量：约10万条交易记录
受众：公司管理层
输出：PDF报告 + HTML仪表板

分析步骤

  1. 数据准备：加载和清洗销售数据，处理缺失值和异常值
  2. 探索性分析：使用Seaborn快速了解数据分布和相关性
  3. 趋势分析：使用Matplotlib绑制销售趋势折线图
  4. 结构分析：使用饼图和柱状图分析产品和区域结构
  5. 交互仪表板：使用Plotly创建可交互的分析仪表板
  6. 报告输出：整合所有图表，输出分析报告

二、数据准备与探索

# 综合案例：电商销售数据可视化分析 # 第一步：数据准备与探索 import pandas as pd import numpy as np import matplotlib.pyplot as plt import seaborn as sns import plotly.express as px import plotly.graph_objects as go from plotly.subplots import make_subplots # 设置中文字体和样式 plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS'] plt.rcParams['axes.unicode_minus'] = False sns.set_theme(style='whitegrid') # 生成模拟数据 np.random.seed(42) # 月度销售数据 months = pd.date_range('2024-01', '2024-06', freq='M').strftime('%Y-%m').tolist() sales_data = pd.DataFrame({ '月份': months * 2, '年份': ['2023']*6 + ['2024']*6, '销售额': [100, 115, 128, 135, 148, 162, 120, 145, 162, 178, 195, 210], '订单数': [1200, 1380, 1540, 1620, 1780, 1940, 1440, 1740, 1940, 2136, 2340, 2520], '客单价': [83.3, 83.3, 83.1, 83.3, 83.1, 83.5, 83.3, 83.3, 83.5, 83.3, 83.3, 83.3] }) # 区域销售数据 regions = ['华东', '华南', '华北', '西部'] region_data = pd.DataFrame({ '区域': regions, '线上销售': [180, 145, 120, 95], '线下销售': [120, 95, 85, 75], '增长率': [18.5, 15.2, 12.8, 8.5] }) # 产品销售数据 products = ['电子产品', '服装', '食品', '家居', '美妆'] product_data = pd.DataFrame({ '产品': products, '销售额': [156, 128, 95, 82, 68], '毛利率': [25, 35, 18, 22, 42] }) # 用户数据 user_data = pd.DataFrame({ '用户等级': ['普通用户', 'VIP用户', 'SVIP用户'], '用户数': [45000, 28000, 7000], '贡献销售额': [35, 42, 23] }) print("数据准备完成！") print(f"销售数据：{len(sales_data)} 条记录") print(f"区域数据：{len(region_data)} 个区域") print(f"产品数据：{len(product_data)} 个品类")

数据探索要点

在开始可视化之前，先使用describe()、info()等方法了解数据基本情况，检查缺失值和异常值。这一步决定了后续分析的准确性。

三、销售趋势分析（Matplotlib）

# 第二步：销售趋势分析 - 使用Matplotlib import matplotlib.pyplot as plt import numpy as np # 准备数据 months_2023 = sales_data[sales_data['年份']=='2023']['月份'].str.slice(5).tolist() months_2024 = sales_data[sales_data['年份']=='2024']['月份'].str.slice(5).tolist() sales_2023 = sales_data[sales_data['年份']=='2023']['销售额'].tolist() sales_2024 = sales_data[sales_data['年份']=='2024']['销售额'].tolist() # 创建专业商务图表 fig, axes = plt.subplots(1, 2, figsize=(14, 5)) # 左图：销售趋势对比 ax1 = axes[0] ax1.plot(months_2023, sales_2023, marker='o', color='#94a3b8', linewidth=2, markersize=6, label='2023年') ax1.plot(months_2024, sales_2024, marker='o', color='#2563eb', linewidth=3, markersize=8, label='2024年') # 添加数据标签 ax1.annotate(f'{sales_2024[-1]}万', xy=(5, sales_2024[-1]), xytext=(10, 0), textcoords='offset points', fontsize=12, fontweight='bold', color='#2563eb') # 计算并标注增长率 growth = (sales_2024[-1] - sales_2023[-1]) / sales_2023[-1] * 100 ax1.text(0.02, 0.98, f'同比增长: {growth:.1f}%', transform=ax1.transAxes, fontsize=11, verticalalignment='top', bbox=dict(boxstyle='round', facecolor='#f0fdf4', edgecolor='#10b981')) ax1.set_title('月度销售趋势对比', fontsize=14, fontweight='bold', pad=15) ax1.set_ylabel('销售额（万元）', fontsize=11) ax1.legend(loc='upper left', frameon=False) ax1.spines['top'].set_visible(False) ax1.spines['right'].set_visible(False) ax1.yaxis.grid(True, linestyle='--', alpha=0.3) # 右图：月度增长率 ax2 = axes[1] growth_rates = [(s2-s1)/s1*100 for s1, s2 in zip(sales_2023, sales_2024)] colors = ['#10b981' if g > 15 else '#f59e0b' for g in growth_rates] bars = ax2.bar(months_2024, growth_rates, color=colors, width=0.6) ax2.axhline(y=np.mean(growth_rates), color='#ef4444', linestyle='--', linewidth=2, label=f'平均增长率 {np.mean(growth_rates):.1f}%') ax2.set_title('月度同比增长率', fontsize=14, fontweight='bold', pad=15) ax2.set_ylabel('增长率（%）', fontsize=11) ax2.legend(frameon=False) ax2.spines['top'].set_visible(False) ax2.spines['right'].set_visible(False) plt.suptitle('销售趋势分析报告', fontsize=16, fontweight='bold', y=1.02) plt.tight_layout() plt.savefig('sales_trend_analysis.png', dpi=150, bbox_inches='tight') plt.show() print("\n趋势分析洞察：") print(f"1. 2024年上半年销售额持续增长，6月达到{sales_2024[-1]}万元") print(f"2. 同比增长率平均{np.mean(growth_rates):.1f}%，增长态势良好") print(f"3. 2月增长率最高，达{max(growth_rates):.1f}%")

小贴士：趋势分析图表要突出关键信息：增长率、转折点、异常值。使用颜色区分历史数据和当前数据，添加关键数据标注。

四、区域与产品分析（Seaborn）

# 第三步：区域与产品分析 - 使用Seaborn import seaborn as sns import matplotlib.pyplot as plt # 创建综合分析图表 fig = plt.figure(figsize=(16, 10)) # 1. 区域销售对比（分组柱状图） ax1 = fig.add_subplot(2, 2, 1) region_long = region_data.melt(id_vars=['区域', '增长率'], var_name='渠道', value_name='销售额') region_long['渠道'] = region_long['渠道'].str.replace('销售', '') sns.barplot(data=region_long, x='区域', y='销售额', hue='渠道', palette={'线上': '#2563eb', '线下': '#10b981'}, ax=ax1) ax1.set_title('各区域渠道销售对比', fontsize=14, fontweight='bold') ax1.legend(title='', frameon=False) # 2. 区域增长率（柱状图+数值标注） ax2 = fig.add_subplot(2, 2, 2) colors = ['#10b981' if g > 15 else '#f59e0b' for g in region_data['增长率']] bars = ax2.bar(region_data['区域'], region_data['增长率'], color=colors) for bar, val in zip(bars, region_data['增长率']): ax2.annotate(f'{val}%', xy=(bar.get_x() + bar.get_width()/2, bar.get_height()), xytext=(0, 5), textcoords='offset points', ha='center', fontsize=11, fontweight='bold') ax2.set_title('各区域同比增长率', fontsize=14, fontweight='bold') ax2.set_ylabel('增长率（%）') ax2.spines['top'].set_visible(False) ax2.spines['right'].set_visible(False) # 3. 产品销售结构（环形图） ax3 = fig.add_subplot(2, 2, 3) colors = ['#2563eb', '#3b82f6', '#60a5fa', '#93c5fd', '#bfdbfe'] wedges, texts, autotexts = ax3.pie(product_data['销售额'], labels=product_data['产品'], colors=colors, autopct='%1.1f%%', wedgeprops=dict(width=0.5), startangle=90) ax3.set_title('产品销售结构', fontsize=14, fontweight='bold') # 4. 产品销售额与毛利率（气泡图） ax4 = fig.add_subplot(2, 2, 4) scatter = ax4.scatter(product_data['销售额'], product_data['毛利率'], s=product_data['销售额']*2, c=product_data['毛利率'], cmap='RdYlGn', alpha=0.6, edgecolors='white', linewidths=2) for i, product in enumerate(product_data['产品']): ax4.annotate(product, (product_data['销售额'].iloc[i], product_data['毛利率'].iloc[i]), textcoords='offset points', xytext=(0, 10), ha='center') ax4.set_xlabel('销售额（万元）') ax4.set_ylabel('毛利率（%）') ax4.set_title('产品销售额与毛利率分析', fontsize=14, fontweight='bold') ax4.spines['top'].set_visible(False) ax4.spines['right'].set_visible(False) plt.suptitle('区域与产品综合分析', fontsize=16, fontweight='bold', y=1.02) plt.tight_layout() plt.savefig('region_product_analysis.png', dpi=150, bbox_inches='tight') plt.show() print("\n区域与产品分析洞察：") print("1. 华东地区销售额最高，线上线下均领先") print("2. 美妆品类毛利率最高(42%)，但销售额较低") print("3. 电子产品是销售主力，占比约30%")

五、交互式仪表板（Plotly）

# 第四步：创建交互式仪表板 - 使用Plotly import plotly.graph_objects as go from plotly.subplots import make_subplots # 创建综合仪表板 fig = make_subplots( rows=2, cols=3, subplot_titles=( '销售趋势对比', '区域销售分析', '产品结构占比', '用户等级分布', '渠道销售对比', '关键指标' ), specs=[ [{'type': 'scatter'}, {'type': 'bar'}, {'type': 'pie'}], [{'type': 'pie'}, {'type': 'bar'}, {'type': 'indicator'}] ] ) # 1. 销售趋势对比 fig.add_trace(go.Scatter( x=months_2023, y=sales_2023, name='2023年', mode='lines+markers', line=dict(color='#94a3b8', width=2, dash='dot') ), row=1, col=1) fig.add_trace(go.Scatter( x=months_2024, y=sales_2024, name='2024年', mode='lines+markers', line=dict(color='#2563eb', width=3) ), row=1, col=1) # 2. 区域销售分析 fig.add_trace(go.Bar( name='线上', x=region_data['区域'], y=region_data['线上销售'], marker_color='#2563eb' ), row=1, col=2) fig.add_trace(go.Bar( name='线下', x=region_data['区域'], y=region_data['线下销售'], marker_color='#10b981' ), row=1, col=2) # 3. 产品结构占比 fig.add_trace(go.Pie( labels=product_data['产品'], values=product_data['销售额'], marker_colors=['#2563eb', '#3b82f6', '#60a5fa', '#93c5fd', '#bfdbfe'], hole=0.4 ), row=1, col=3) # 4. 用户等级分布 fig.add_trace(go.Pie( labels=user_data['用户等级'], values=user_data['贡献销售额'], marker_colors=['#94a3b8', '#3b82f6', '#f59e0b'], hole=0.4 ), row=2, col=1) # 5. 渠道销售对比 channel_total = [region_data['线上销售'].sum(), region_data['线下销售'].sum()] fig.add_trace(go.Bar( x=['线上渠道', '线下渠道'], y=channel_total, marker_color=['#2563eb', '#10b981'], text=channel_total, textposition='outside' ), row=2, col=2) # 6. 关键指标 total_sales = sum(sales_2024) fig.add_trace(go.Indicator( mode='number+delta', value=total_sales, title={'text': '上半年总销售额（万元）'}, delta={'reference': sum(sales_2023), 'relative': True} ), row=2, col=3) # 更新布局 fig.update_layout( title_text='2024年上半年销售数据综合仪表板', title_font_size=20, title_x=0.5, showlegend=True, legend=dict(orientation='h', yanchor='bottom', y=1.02), template='plotly_white', height=700, barmode='group' ) # 显示仪表板 fig.show() # 导出HTML fig.write_html('sales_dashboard.html') print("\n交互式仪表板已生成并保存为 sales_dashboard.html") print("可在浏览器中打开查看完整交互功能")

仪表板设计要点

交互式仪表板要遵循"概览优先，细节可探"原则。主视图展示关键指标，用户可通过交互深入探索。合理使用颜色区分数据维度，保持整体风格一致。

# 生成分析报告摘要 report_summary = """ ======================================== 2024年上半年销售数据分析报告摘要 ======================================== 一、整体表现 - 上半年总销售额：1010万元，同比增长29.6% - 订单总量：12,100单，同比增长32.5% - 客单价稳定在83元左右 二、区域分析 - 华东地区表现最佳，销售额占比35% - 西部地区增长最慢(8.5%)，建议加大投入 - 线上渠道整体表现优于线下 三、产品分析 - 电子产品销售额最高，但毛利率偏低(25%) - 美妆品类毛利率最高(42%)，建议重点推广 - 食品类增长稳定，是稳定的现金流来源 四、用户分析 - VIP用户仅占35%，却贡献42%销售额 - SVIP用户贡献23%销售额，价值最高 - 建议加强会员运营，提升用户等级 五、行动建议 1. 加大西部地区市场推广投入 2. 重点发展美妆品类，提升整体毛利 3. 优化会员权益，促进用户升级 4. 持续投入线上渠道建设 5. 关注高毛利产品的营销推广 """ print(report_summary) # 保存报告 with open('sales_analysis_report.txt', 'w', encoding='utf-8') as f: f.write(report_summary) print("\n报告已保存为 sales_analysis_report.txt")
