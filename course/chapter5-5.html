综合运用ABC分析、利润分析、关联规则和库存分析方法解决实际业务问题

案例：某连锁超市销售运营综合分析

背景：某连锁超市拥有50家门店，年销售额约5亿元。近期发现销售额增长放缓，利润率下降，库存周转效率降低。管理层希望通过数据分析找出问题并提出优化方案。

数据范围：最近12个月的销售数据、库存数据、成本数据

一、数据准备与概览

# ======================================== # 综合案例分析 - 数据准备 # ======================================== import pandas as pd import numpy as np import matplotlib.pyplot as plt plt.rcParams['font.sans-serif'] = ['SimHei'] plt.rcParams['axes.unicode_minus'] = False # 1. 销售数据 sales_data = pd.DataFrame({ '月份': ['1月', '2月', '3月', '4月', '5月', '6月', '7月', '8月', '9月', '10月', '11月', '12月'], '营业收入': [4200, 3800, 4500, 4100, 4300, 4000, 3900, 4100, 4400, 4600, 4800, 5200], '营业成本': [2800, 2600, 3000, 2800, 2900, 2700, 2700, 2800, 2900, 3100, 3200, 3500], '期间费用': [700, 650, 720, 680, 700, 680, 660, 690, 710, 730, 750, 800] }) # 2. 产品数据 product_data = pd.DataFrame({ '产品ID': ['P001', 'P002', 'P003', 'P004', 'P005', 'P006', 'P007', 'P008'], '产品名称': ['生鲜食品', '饮料', '零食', '日用品', '乳制品', '调味品', '酒类', '冷冻食品'], '年销售额': [15000, 12000, 8000, 6000, 5000, 2500, 1200, 800], '平均库存': [1500, 1000, 1200, 2000, 800, 600, 400, 300], '毛利率': [0.18, 0.25, 0.35, 0.22, 0.20, 0.28, 0.40, 0.15] }) # 3. 数据概览 print("=" * 60) print("数据概览") print("=" * 60) print(f"年度总销售额: {sales_data['营业收入'].sum()} 万元") print(f"年度总成本: {sales_data['营业成本'].sum()} 万元") print(f"产品数量: {len(product_data)} 个品类") print(f"门店数量: 50 家")

二、分析步骤一：利润分析

1利润率趋势分析

# 利润分析 sales_data['毛利'] = sales_data['营业收入'] - sales_data['营业成本'] sales_data['净利润'] = sales_data['毛利'] - sales_data['期间费用'] sales_data['毛利率'] = sales_data['毛利'] / sales_data['营业收入'] sales_data['净利率'] = sales_data['净利润'] / sales_data['营业收入'] print("月度利润率分析") print(sales_data[['月份', '营业收入', '毛利率', '净利率']].round(3)) # 年度汇总 annual_revenue = sales_data['营业收入'].sum() annual_gross = sales_data['毛利'].sum() annual_net = sales_data['净利润'].sum() print(f"\n年度毛利率: {annual_gross/annual_revenue*100:.1f}%") print(f"年度净利率: {annual_net/annual_revenue*100:.1f}%")
分析发现

毛利率稳定在33%左右，净利率约16%。费用率约17%，处于行业正常水平。利润率整体健康，但增长空间有限。

三、分析步骤二：ABC分类分析

2产品ABC分类

# ABC分类分析 product_data = product_data.sort_values('年销售额', ascending=False).reset_index(drop=True) total_sales = product_data['年销售额'].sum() product_data['销售额占比'] = product_data['年销售额'] / total_sales product_data['累计占比'] = product_data['销售额占比'].cumsum() def classify(x): if x <= 0.8: return 'A' elif x <= 0.95: return 'B' else: return 'C' product_data['ABC类别'] = product_data['累计占比'].apply(classify) print("产品ABC分类结果") print(product_data[['产品名称', '年销售额', '销售额占比', 'ABC类别']]) # 分类统计 abc_summary = product_data.groupby('ABC类别').agg({ '产品ID': 'count', '年销售额': 'sum' }) abc_summary['品种占比'] = abc_summary['产品ID'] / len(product_data) * 100 abc_summary['销售占比'] = abc_summary['年销售额'] / total_sales * 100 print("\nABC分类统计") print(abc_summary.round(1))
分析发现

A类产品（生鲜食品、饮料、零食）占品种37.5%，贡献销售额77.5%。应重点管理A类产品的库存和促销。

四、分析步骤三：库存周转分析

3库存周转效率分析

# 库存周转分析 from scipy.stats import norm # 计算各产品库存周转率 product_data['库存周转率'] = product_data['年销售额'] / product_data['平均库存'] product_data['周转天数'] = 365 / product_data['库存周转率'] print("产品库存周转分析") print(product_data[['产品名称', '库存周转率', '周转天数']].round(1)) # 安全库存计算示例（以生鲜食品为例） avg_daily_demand = 15000 / 365 # 日均需求 demand_std = 20 # 需求标准差 lead_time = 2 # 提前期 service_level = 0.95 z_value = norm.ppf(service_level) safety_stock = z_value * demand_std * np.sqrt(lead_time) reorder_point = avg_daily_demand * lead_time + safety_stock print(f"\n生鲜食品安全库存: {safety_stock:.0f} 万元") print(f"补货点: {reorder_point:.0f} 万元")
分析发现

生鲜食品周转最快（10次/年），日用品周转最慢（3次/年）。建议优化日用品库存，减少资金占用。

五、分析步骤四：关联规则分析

4购物篮关联分析

# 关联规则分析（模拟数据） from itertools import combinations # 模拟交易数据 transactions = [ ['生鲜食品', '乳制品'], ['生鲜食品', '饮料'], ['饮料', '零食'], ['生鲜食品', '乳制品', '饮料'], ['日用品', '调味品'], ['生鲜食品', '冷冻食品'], ['饮料', '零食', '酒类'], ['生鲜食品', '乳制品'], ['零食', '饮料'], ['生鲜食品', '饮料', '零食'] ] # 计算关联规则 def find_associations(transactions, min_support=0.2, min_confidence=0.5): n = len(transactions) rules = [] items = set() for t in transactions: items.update(t) for item1 in items: for item2 in items: if item1 != item2: both = sum(1 for t in transactions if item1 in t and item2 in t) first = sum(1 for t in transactions if item1 in t) second = sum(1 for t in transactions if item2 in t) support = both / n confidence = both / first if first > 0 else 0 lift = confidence / (second / n) if second > 0 else 0 if support >= min_support and confidence >= min_confidence and lift > 1: rules.append({ '规则': f"{item1} → {item2}", '支持度': round(support, 2), '置信度': round(confidence, 2), '提升度': round(lift, 2) }) return pd.DataFrame(rules).sort_values('提升度', ascending=False) rules_df = find_associations(transactions) print("发现的关联规则") print(rules_df)
分析发现

发现强关联：生鲜食品→乳制品、饮料→零食。建议将这些商品陈列在相邻位置，并设计组合促销活动。

七、完整代码整合

# ======================================== # 销售运营综合分析 - 完整代码 # ======================================== import pandas as pd import numpy as np from scipy.stats import norm import matplotlib.pyplot as plt plt.rcParams['font.sans-serif'] = ['SimHei'] plt.rcParams['axes.unicode_minus'] = False class SalesAnalyzer: """销售运营综合分析类""" def __init__(self, sales_data, product_data): self.sales_data = sales_data self.product_data = product_data def profit_analysis(self): """利润分析""" df = self.sales_data.copy() df['毛利'] = df['营业收入'] - df['营业成本'] df['净利润'] = df['毛利'] - df['期间费用'] df['毛利率'] = df['毛利'] / df['营业收入'] df['净利率'] = df['净利润'] / df['营业收入'] return df def abc_analysis(self): """ABC分类分析""" df = self.product_data.sort_values('年销售额', ascending=False).copy() total = df['年销售额'].sum() df['销售额占比'] = df['年销售额'] / total df['累计占比'] = df['销售额占比'].cumsum() df['ABC类别'] = df['累计占比'].apply( lambda x: 'A' if x <= 0.8 else ('B' if x <= 0.95 else 'C') ) return df def inventory_analysis(self): """库存周转分析""" df = self.product_data.copy() df['周转率'] = df['年销售额'] / df['平均库存'] df['周转天数'] = 365 / df['周转率'] return df def generate_report(self): """生成综合报告""" profit = self.profit_analysis() abc = self.abc_analysis() inventory = self.inventory_analysis() report = { '年度营收': profit['营业收入'].sum(), '年度毛利率': profit['毛利'].sum() / profit['营业收入'].sum(), '年度净利率': profit['净利润'].sum() / profit['营业收入'].sum(), 'A类产品数': len(abc[abc['ABC类别'] == 'A']), '平均周转率': inventory['周转率'].mean() } return report # 使用示例 analyzer = SalesAnalyzer(sales_data, product_data) report = analyzer.generate_report() print("=" * 60) print("综合分析报告") print("=" * 60) for key, value in report.items(): if '率' in key: print(f"{key}: {value*100:.1f}%") else: print(f"{key}: {value}")
