一、Plotly简介

什么是Plotly？

Plotly是一个强大的Python交互式可视化库，可以创建具有缩放、平移、悬停提示等交互功能的图表。它支持导出HTML文件，方便在网页中嵌入和分享。

交互性强

支持缩放、平移、悬停提示、点击事件等

易于分享

导出HTML，无需后端即可展示

美观现代

默认样式精美，适合现代报告

Plotly vs Matplotlib

特性 Matplotlib Plotly
交互性 静态图表 原生交互
学习曲线 较陡峭 平缓易学
输出格式 PNG/PDF/SVG HTML/图片
适用场景 学术论文、印刷报告 网页展示、演示报告
# 安装和导入Plotly # pip install plotly import plotly.express as px import plotly.graph_objects as go import pandas as pd import numpy as np # Plotly有两个主要模块： # 1. plotly.express (px): 高级API，快速绑图 # 2. plotly.graph_objects (go): 底层API，精细控制 print("Plotly版本:", px.__version__) # 在Jupyter Notebook中显示图表 # fig.show() 会自动渲染交互式图表 # 导出为HTML文件 # fig.write_html("chart.html")

小贴士：推荐优先使用Plotly Express (px)进行快速绑图，它提供了类似Seaborn的简洁API。需要更精细控制时再使用graph_objects (go)。

二、交互式折线图

基础折线图

# 交互式折线图示例 import plotly.express as px import pandas as pd # 准备数据 df = pd.DataFrame({ '月份': ['1月', '2月', '3月', '4月', '5月', '6月'], '销售额': [120, 145, 162, 178, 195, 210], '成本': [80, 95, 105, 115, 125, 135] }) # 创建折线图 fig = px.line(df, x='月份', y='销售额', title='月度销售趋势', markers=True, line_shape='linear') # 自定义样式 fig.update_traces( line=dict(color='#2563eb', width=3), marker=dict(size=10, color='#2563eb') ) fig.update_layout( title_font_size=20, title_font_weight='bold', xaxis_title='月份', yaxis_title='销售额（万元）', hovermode='x unified', # 统一悬停模式 template='plotly_white' ) # 显示图表 fig.show() # 导出HTML # fig.write_html("sales_trend.html")

多系列折线图

# 多系列折线图：销售额与成本对比 import plotly.express as px import pandas as pd # 准备数据（长格式） df_long = pd.DataFrame({ '月份': ['1月', '2月', '3月', '4月', '5月', '6月'] * 2, '指标': ['销售额']*6 + ['成本']*6, '金额': [120, 145, 162, 178, 195, 210, 80, 95, 105, 115, 125, 135] }) # 创建多系列折线图 fig = px.line(df_long, x='月份', y='金额', color='指标', title='销售额与成本趋势对比', markers=True, color_discrete_map={'销售额': '#2563eb', '成本': '#ef4444'}) # 添加利润线（使用graph_objects） import plotly.graph_objects as go profit = [120-80, 145-95, 162-105, 178-115, 195-125, 210-135] fig.add_trace(go.Scatter( x=df['月份'].tolist(), y=profit, mode='lines+markers', name='利润', line=dict(color='#10b981', width=3, dash='dot') )) fig.update_layout( title_font_size=18, legend_title_text='', hovermode='x unified', template='plotly_white' ) fig.show()

三、交互式柱状图

基础柱状图

# 交互式柱状图 import plotly.express as px import pandas as pd # 准备数据 df = pd.DataFrame({ '产品': ['产品A', '产品B', '产品C', '产品D', '产品E'], '销售额': [156, 128, 142, 98, 175], '增长率': [12.5, -5.2, 8.3, -2.1, 15.8] }) # 创建柱状图 fig = px.bar(df, x='产品', y='销售额', title='各产品销售额对比', color='增长率', color_continuous_scale=['#ef4444', '#fbbf24', '#10b981'], text='销售额') # 自定义样式 fig.update_traces( texttemplate='%{text}万', textposition='outside', marker_line_width=0 ) fig.update_layout( title_font_size=18, xaxis_title='产品', yaxis_title='销售额（万元）', coloraxis_colorbar=dict(title='增长率(%)'), template='plotly_white', yaxis_range=[0, max(df['销售额']) * 1.2] ) fig.show()

分组与堆叠柱状图

# 分组柱状图 vs 堆叠柱状图 import plotly.express as px import plotly.graph_objects as go from plotly.subplots import make_subplots import pandas as pd # 准备数据 df = pd.DataFrame({ '区域': ['华东', '华南', '华北', '西部'], '线上': [180, 145, 120, 95], '线下': [120, 95, 85, 75] }) # 创建子图 fig = make_subplots(rows=1, cols=2, subplot_titles=('分组柱状图', '堆叠柱状图')) # 分组柱状图 fig.add_trace(go.Bar(name='线上', x=df['区域'], y=df['线上'], marker_color='#2563eb'), row=1, col=1) fig.add_trace(go.Bar(name='线下', x=df['区域'], y=df['线下'], marker_color='#10b981'), row=1, col=1) # 堆叠柱状图 fig.add_trace(go.Bar(name='线上', x=df['区域'], y=df['线上'], marker_color='#2563eb'), row=1, col=2) fig.add_trace(go.Bar(name='线下', x=df['区域'], y=df['线下'], marker_color='#10b981'), row=1, col=2) fig.update_layout( title_text='线上线下销售对比', title_font_size=18, barmode='group', template='plotly_white', showlegend=True ) # 第二个子图设置为堆叠模式 fig.update_layout(barmode='group') fig.update_xaxes(title_text='区域') fig.update_yaxes(title_text='销售额（万元）') fig.show()
知识点测验

在Plotly Express中，如何创建分组柱状图？

设置 barmode='stacked'
使用 color 参数指定分组变量
设置 group=True
使用多个 px.bar() 函数

解析：在Plotly Express中，使用color参数指定分组变量即可自动生成分组柱状图。通过fig.update_layout(barmode='group'或'stack')可以切换分组或堆叠模式。

四、交互式散点图与气泡图

散点图

# 交互式散点图 import plotly.express as px import pandas as pd import numpy as np # 生成模拟数据 np.random.seed(42) n = 100 df = pd.DataFrame({ '广告投入': np.random.uniform(10, 100, n), '销售额': np.random.uniform(50, 300, n), '产品类别': np.random.choice(['电子产品', '服装', '食品'], n), '利润率': np.random.uniform(10, 40, n) }) # 添加相关性 df['销售额'] = df['广告投入'] * 2.5 + np.random.normal(0, 30, n) # 创建散点图 fig = px.scatter(df, x='广告投入', y='销售额', color='产品类别', size='利润率', hover_data={'利润率': ':.1f'}, title='广告投入与销售额关系分析', color_discrete_map={ '电子产品': '#2563eb', '服装': '#10b981', '食品': '#f59e0b' }) # 添加趋势线 fig.update_layout( title_font_size=18, xaxis_title='广告投入（万元）', yaxis_title='销售额（万元）', template='plotly_white', legend_title_text='产品类别' ) fig.show()

地理散点图

# 地理散点图：区域销售分布 import plotly.express as px import pandas as pd # 准备城市销售数据 df = pd.DataFrame({ '城市': ['北京', '上海', '广州', '深圳', '杭州', '成都'], '销售额': [280, 320, 180, 165, 145, 120], 'lat': [39.9, 31.2, 23.1, 22.5, 30.3, 30.7], 'lon': [116.4, 121.5, 113.3, 114.1, 120.2, 104.1] }) # 创建地图散点图 fig = px.scatter_mapbox(df, lat='lat', lon='lon', size='销售额', color='销售额', hover_name='城市', hover_data={'销售额': True}, color_continuous_scale='Blues', size_max=30, zoom=4, title='全国主要城市销售分布') fig.update_layout( mapbox_style='open-street-map', title_font_size=18, margin=dict(l=0, r=0, t=50, b=0) ) fig.show()

小贴士：Plotly的散点图支持多种hover模式。使用hover_data参数可以自定义悬停时显示的信息，hover_name指定悬停标题。

五、交互式饼图与环形图

# 交互式饼图与环形图 import plotly.graph_objects as go from plotly.subplots import make_subplots # 准备数据 labels = ['电子产品', '服装', '食品', '家居', '其他'] values = [35, 25, 18, 12, 10] colors = ['#2563eb', '#3b82f6', '#60a5fa', '#93c5fd', '#bfdbfe'] # 创建子图 fig = make_subplots(rows=1, cols=2, specs=[[{'type':'pie'}, {'type':'pie'}]], subplot_titles=['饼图', '环形图']) # 饼图 fig.add_trace(go.Pie( labels=labels, values=values, marker_colors=colors, textinfo='label+percent', textposition='outside', hole=0 ), row=1, col=1) # 环形图 fig.add_trace(go.Pie( labels=labels, values=values, marker_colors=colors, textinfo='label+percent', textposition='outside', hole=0.4 # 环形图的关键参数 ), row=1, col=2) fig.update_layout( title_text='产品销售占比分析', title_font_size=18, showlegend=False, template='plotly_white' ) fig.show()

注意：Plotly Express不直接支持饼图，需要使用plotly.graph_objects.Pie。饼图扇区过多时，悬停信息会显得杂乱，建议合并小比例为"其他"。

六、子图与仪表板布局

创建多子图仪表板

# 综合仪表板示例 import plotly.graph_objects as go from plotly.subplots import make_subplots import pandas as pd import numpy as np # 准备数据 np.random.seed(42) months = ['1月', '2月', '3月', '4月', '5月', '6月'] sales = [120, 145, 162, 178, 195, 210] products = ['产品A', '产品B', '产品C', '产品D'] product_sales = [156, 128, 142, 98] # 创建2x2子图 fig = make_subplots( rows=2, cols=2, subplot_titles=('销售趋势', '产品占比', '区域对比', '渠道分析'), specs=[[{'type': 'scatter'}, {'type': 'pie'}], [{'type': 'bar'}, {'type': 'bar'}]] ) # 1. 销售趋势折线图 fig.add_trace(go.Scatter( x=months, y=sales, mode='lines+markers', name='销售额', line=dict(color='#2563eb', width=3) ), row=1, col=1) # 2. 产品占比饼图 fig.add_trace(go.Pie( labels=products, values=product_sales, marker_colors=['#2563eb', '#3b82f6', '#60a5fa', '#93c5fd'], hole=0.4 ), row=1, col=2) # 3. 区域对比柱状图 regions = ['华东', '华南', '华北', '西部'] region_sales = [180, 145, 120, 95] fig.add_trace(go.Bar( x=regions, y=region_sales, marker_color='#2563eb' ), row=2, col=1) # 4. 渠道分析柱状图 channels = ['线上', '线下'] channel_sales = [320, 220] fig.add_trace(go.Bar( x=channels, y=channel_sales, marker_color=['#2563eb', '#10b981'] ), row=2, col=2) # 更新布局 fig.update_layout( title_text='销售数据综合仪表板', title_font_size=20, showlegend=False, template='plotly_white', height=700 ) # 更新坐标轴标题 fig.update_xaxes(title_text='月份', row=1, col=1) fig.update_yaxes(title_text='销售额（万元）', row=1, col=1) fig.update_xaxes(title_text='区域', row=2, col=1) fig.update_yaxes(title_text='销售额（万元）', row=2, col=1) fig.show() # 导出HTML仪表板 # fig.write_html("sales_dashboard.html")

小贴士：使用make_subplots创建仪表板时，通过specs参数指定每个子图的类型。支持混合不同类型的图表，如折线图、饼图、柱状图等。

七、真实商业案例：交互式销售仪表板

案例背景

某电商公司需要创建一个可交互的销售数据仪表板，支持管理层在线查看和探索数据，要求包含趋势分析、区域对比、产品结构等多维度视图。

# 完整的交互式销售仪表板 import plotly.graph_objects as go from plotly.subplots import make_subplots import pandas as pd import numpy as np # 生成模拟数据 np.random.seed(42) # 月度数据 months = pd.date_range('2024-01', '2024-06', freq='M').strftime('%Y-%m').tolist() sales_2024 = [120, 145, 162, 178, 195, 210] sales_2023 = [100, 115, 128, 135, 148, 162] # 区域数据 regions = ['华东', '华南', '华北', '西部'] region_online = [180, 145, 120, 95] region_offline = [120, 95, 85, 75] # 产品数据 products = ['电子产品', '服装', '食品', '家居', '美妆'] product_sales = [156, 128, 95, 82, 68] # 创建仪表板 fig = make_subplots( rows=2, cols=3, subplot_titles=( '销售趋势对比', '区域渠道分析', '产品结构占比', '月度增长率', 'TOP5产品销售', '关键指标卡片' ), specs=[ [{'type': 'scatter'}, {'type': 'bar'}, {'type': 'pie'}], [{'type': 'bar'}, {'type': 'bar'}, {'type': 'indicator'}] ], row_heights=[0.5, 0.5] ) # 1. 销售趋势对比 fig.add_trace(go.Scatter( x=months, y=sales_2024, name='2024年', mode='lines+markers', line=dict(color='#2563eb', width=3) ), row=1, col=1) fig.add_trace(go.Scatter( x=months, y=sales_2023, name='2023年', mode='lines+markers', line=dict(color='#94a3b8', width=2, dash='dot') ), row=1, col=1) # 2. 区域渠道分析 fig.add_trace(go.Bar( name='线上', x=regions, y=region_online, marker_color='#2563eb' ), row=1, col=2) fig.add_trace(go.Bar( name='线下', x=regions, y=region_offline, marker_color='#10b981' ), row=1, col=2) # 3. 产品结构占比 fig.add_trace(go.Pie( labels=products, values=product_sales, marker_colors=['#2563eb', '#3b82f6', '#60a5fa', '#93c5fd', '#bfdbfe'], hole=0.4, textinfo='percent' ), row=1, col=3) # 4. 月度增长率 growth_rates = [(s2-s1)/s1*100 for s1, s2 in zip(sales_2023, sales_2024)] colors = ['#10b981' if g > 0 else '#ef4444' for g in growth_rates] fig.add_trace(go.Bar( x=months, y=growth_rates, marker_color=colors ), row=2, col=1) # 5. TOP5产品销售 fig.add_trace(go.Bar( x=products, y=product_sales, marker_color='#2563eb', text=product_sales, textposition='outside' ), row=2, col=2) # 6. 关键指标卡片 total_sales = sum(sales_2024) avg_growth = np.mean(growth_rates) fig.add_trace(go.Indicator( mode='number+delta', value=total_sales, title={'text': '总销售额（万元）'}, delta={'reference': sum(sales_2023), 'relative': True}, domain={'row': 1, 'column': 2} ), row=2, col=3) # 更新布局 fig.update_layout( title_text='电商销售数据交互式仪表板', title_font_size=22, title_x=0.5, showlegend=True, legend=dict(orientation='h', yanchor='bottom', y=1.02), template='plotly_white', height=800, barmode='group' ) # 更新坐标轴 fig.update_yaxes(title_text='销售额（万元）', row=1, col=1) fig.update_yaxes(title_text='销售额（万元）', row=1, col=2) fig.update_yaxes(title_text='增长率(%)', row=2, col=1) fig.update_yaxes(title_text='销售额（万元）', row=2, col=2) fig.show() # 导出为HTML文件 # fig.write_html("sales_dashboard.html") print("仪表板已生成！可导出为HTML文件分享。")

如何将Plotly图表导出为HTML文件？
