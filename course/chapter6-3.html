一、Seaborn简介与优势

什么是Seaborn？

Seaborn是基于Matplotlib构建的Python数据可视化库，专注于统计图表的绑制。它提供了更高级的接口，让创建美观的统计图表变得简单。

Seaborn优势

  • 内置精美主题和配色方案
  • 自动处理DataFrame数据
  • 一键绑制统计图表
  • 支持多维度数据映射
  • 默认输出更美观

与Matplotlib对比

  • Matplotlib更底层，控制更精细
  • Seaborn更适合快速探索
  • Seaborn基于Matplotlib构建
  • 两者可以混合使用
  • Seaborn默认样式更现代
# 安装和导入Seaborn # pip install seaborn import seaborn as sns import matplotlib.pyplot as plt import pandas as pd import numpy as np # 设置中文字体 plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS'] plt.rcParams['axes.unicode_minus'] = False # 设置Seaborn主题 sns.set_theme(style='whitegrid') # 可选: darkgrid, white, dark, ticks # 设置调色板 sns.set_palette('Blues') print("Seaborn版本:", sns.__version__) print("可用主题:", ['darkgrid', 'whitegrid', 'dark', 'white', 'ticks'])

二、分布图：直方图与密度图

单变量分布分析

分布图用于展示单个变量的数据分布情况，是数据探索的重要工具。

# 分布图示例：用户消费金额分析 import seaborn as sns import matplotlib.pyplot as plt import numpy as np import pandas as pd # 生成模拟数据 np.random.seed(42) n = 1000 data = pd.DataFrame({ '消费金额': np.random.lognormal(mean=4, sigma=0.5, size=n), '用户等级': np.random.choice(['普通', 'VIP', 'SVIP'], size=n, p=[0.6, 0.3, 0.1]) }) # 创建图表 fig, axes = plt.subplots(1, 3, figsize=(15, 5)) # 1. 直方图 sns.histplot(data=data, x='消费金额', bins=30, color='#2563eb', ax=axes[0]) axes[0].set_title('直方图：消费金额分布', fontsize=14, fontweight='bold') axes[0].set_xlabel('消费金额（元）') # 2. 核密度估计图 sns.kdeplot(data=data, x='消费金额', fill=True, color='#10b981', ax=axes[1]) axes[1].set_title('密度图：消费金额分布', fontsize=14, fontweight='bold') # 3. 组合图（推荐） sns.histplot(data=data, x='消费金额', kde=True, bins=30, color='#2563eb', ax=axes[2]) axes[2].set_title('直方图+密度曲线', fontsize=14, fontweight='bold') plt.tight_layout() plt.show() # 输出统计信息 print("消费金额统计：") print(data['消费金额'].describe().round(2))

分组分布对比

# 按用户等级分组对比分布 import seaborn as sns import matplotlib.pyplot as plt # 设置主题和配色 sns.set_theme(style='whitegrid') colors = ['#94a3b8', '#3b82f6', '#f59e0b'] fig, axes = plt.subplots(1, 2, figsize=(14, 5)) # 方式1：分组密度图 sns.kdeplot(data=data, x='消费金额', hue='用户等级', fill=True, alpha=0.5, palette=colors, ax=axes[0]) axes[0].set_title('各等级用户消费分布对比', fontsize=14, fontweight='bold') # 方式2：小提琴图（结合箱线图和密度图） sns.violinplot(data=data, x='用户等级', y='消费金额', palette=colors, ax=axes[1]) axes[1].set_title('各等级用户消费小提琴图', fontsize=14, fontweight='bold') plt.tight_layout() plt.show()
知识点测验

Seaborn中哪个函数可以同时绘制直方图和密度曲线？

sns.kdeplot()
sns.histplot() 设置 kde=True
sns.distplot()
sns.violinplot()

解析：sns.histplot()函数通过设置kde=True参数，可以在直方图上叠加密度曲线，一步完成两种图表的绑制。

三、热力图：数据相关性分析

相关性矩阵可视化

热力图是展示数据相关性的最佳方式，常用于特征工程和数据探索阶段。

# 热力图：销售数据相关性分析 import seaborn as sns import matplotlib.pyplot as plt import pandas as pd import numpy as np # 创建销售数据 np.random.seed(42) n = 200 sales_data = pd.DataFrame({ '广告投入': np.random.uniform(10, 100, n), '销售额': np.random.uniform(50, 300, n), '客流量': np.random.uniform(100, 500, n), '客单价': np.random.uniform(50, 200, n), '促销力度': np.random.uniform(0, 30, n), '会员占比': np.random.uniform(20, 80, n) }) # 添加相关性（模拟真实数据） sales_data['销售额'] = sales_data['广告投入'] * 2.5 + sales_data['客流量'] * 0.3 + np.random.normal(0, 20, n) sales_data['客单价'] = sales_data['会员占比'] * 1.5 + np.random.normal(0, 15, n) # 计算相关系数矩阵 corr_matrix = sales_data.corr() # 绑制热力图 plt.figure(figsize=(10, 8)) sns.heatmap(corr_matrix, annot=True, # 显示数值 fmt='.2f', # 数值格式 cmap='RdBu_r', # 红蓝配色（反向） center=0, # 颜色中心点 square=True, # 正方形单元格 linewidths=0.5, cbar_kws={'shrink': 0.8}) plt.title('销售指标相关性热力图', fontsize=16, fontweight='bold', pad=20) plt.tight_layout() plt.show() # 输出强相关变量对 print("\n强相关变量对（|r| > 0.5）：") for i in range(len(corr_matrix.columns)): for j in range(i+1, len(corr_matrix.columns)): corr = corr_matrix.iloc[i, j] if abs(corr) > 0.5: print(f" {corr_matrix.columns[i]} - {corr_matrix.columns[j]}: {corr:.3f}")

数据透视热力图

# 透视表热力图：时间-品类销售矩阵 import seaborn as sns import matplotlib.pyplot as plt import pandas as pd import numpy as np # 创建销售数据 np.random.seed(42) months = ['1月', '2月', '3月', '4月', '5月', '6月'] categories = ['电子产品', '服装', '食品', '家居'] # 生成销售矩阵 sales_matrix = pd.DataFrame( np.random.randint(50, 200, size=(len(months), len(categories))), index=months, columns=categories ) # 绑制热力图 plt.figure(figsize=(10, 6)) sns.heatmap(sales_matrix, annot=True, fmt='d', cmap='YlOrRd', # 黄橙红配色 linewidths=0.5, cbar_kws={'label': '销售额（万元）'}) plt.title('月度-品类销售矩阵', fontsize=16, fontweight='bold', pad=20) plt.xlabel('商品品类', fontsize=12) plt.ylabel('月份', fontsize=12) plt.tight_layout() plt.show() print("\n销售数据概览：") print(sales_matrix)

小贴士：热力图的配色选择很重要。相关性矩阵推荐使用双向配色（如RdBu_r），让正相关和负相关都能清晰区分。单一维度的数据推荐使用单向配色（如YlOrRd、Blues）。

四、箱线图与小提琴图

箱线图：数据分布五数概括

箱线图是展示数据分布的经典图表，能同时显示最小值、Q1、中位数、Q3、最大值和异常值。

# 箱线图：各部门薪资分布 import seaborn as sns import matplotlib.pyplot as plt import pandas as pd import numpy as np # 创建薪资数据 np.random.seed(42) departments = ['技术部', '销售部', '市场部', '运营部', '财务部'] salary_data = [] for dept in departments: base = np.random.randint(8000, 15000) n_people = np.random.randint(30, 80) salaries = np.random.normal(base, base*0.2, n_people) for s in salaries: salary_data.append({'部门': dept, '薪资': s}) df = pd.DataFrame(salary_data) # 绑制箱线图 plt.figure(figsize=(12, 6)) sns.boxplot(data=df, x='部门', y='薪资', palette='Blues', width=0.5, flierprops={'marker': 'o', 'markerfacecolor': '#ef4444', 'markersize': 5}) plt.title('各部门薪资分布箱线图', fontsize=16, fontweight='bold', pad=20) plt.ylabel('薪资（元）', fontsize=12) plt.xlabel('部门', fontsize=12) # 添加均值点 means = df.groupby('部门')['薪资'].mean() plt.scatter(range(len(means)), means, color='#f59e0b', s=100, zorder=5, label='均值') plt.legend() plt.tight_layout() plt.show() # 输出统计信息 print("\n各部门薪资统计：") print(df.groupby('部门')['薪资'].agg(['mean', 'median', 'std']).round(0))

小提琴图：分布形态展示

# 小提琴图：更直观的分布展示 import seaborn as sns import matplotlib.pyplot as plt # 使用内置数据集 tips = sns.load_dataset('tips') # 创建对比图 fig, axes = plt.subplots(1, 2, figsize=(14, 6)) # 箱线图 sns.boxplot(data=tips, x='day', y='total_bill', palette='Blues', ax=axes[0]) axes[0].set_title('箱线图', fontsize=14, fontweight='bold') axes[0].set_xlabel('星期') axes[0].set_ylabel('账单金额') # 小提琴图 sns.violinplot(data=tips, x='day', y='total_bill', palette='Blues', inner='box', ax=axes[1]) axes[1].set_title('小提琴图（含箱线图）', fontsize=14, fontweight='bold') axes[1].set_xlabel('星期') axes[1].set_ylabel('账单金额') plt.suptitle('餐厅账单金额分布对比', fontsize=16, fontweight='bold', y=1.02) plt.tight_layout() plt.show()

常见错误：箱线图/小提琴图的数据量过少时，展示效果不佳。建议每组至少有20个以上的数据点，否则应考虑使用散点图或直接展示原始数据。

五、分类图与多维度对比

分类柱状图

# 分类柱状图：多维度销售对比 import seaborn as sns import matplotlib.pyplot asimport pandas as pd import numpy as np # 创建销售数据 np.random.seed(42) regions = ['华东', '华南', '华北', '西部'] channels = ['线上', '线下'] data = [] for region in regions: for channel in channels: sales = np.random.randint(80, 200) data.append({'区域': region, '渠道': channel, '销售额': sales}) df = pd.DataFrame(data) # 创建分组柱状图 plt.figure(figsize=(10, 6)) sns.barplot(data=df, x='区域', y='销售额', hue='渠道', palette={'线上': '#2563eb', '线下': '#10b981'}) plt.title('各区域线上线下销售对比', fontsize=16, fontweight='bold', pad=20) plt.ylabel('销售额（万元）', fontsize=12) plt.xlabel('区域', fontsize=12) plt.legend(title='销售渠道', frameon=False) plt.tight_layout() plt.show()

分面图：多维度拆解

# 分面图：按类别拆解展示 import seaborn as sns import matplotlib.pyplot as plt # 使用内置数据集 penguins = sns.load_dataset('penguins') # 创建分面图 g = sns.FacetGrid(penguins, col='species', row='sex', height=3, aspect=1.2, margin_titles=True) g.map_dataframe(sns.scatterplot, x='bill_length_mm', y='bill_depth_mm', color='#2563eb', alpha=0.6) g.set_axis_labels('喙长 (mm)', '喙深 (mm)') g.set_titles(col_template='{col_name}', row_template='{row_name}') g.fig.suptitle('企鹅喙部特征分面分析', fontsize=14, fontweight='bold', y=1.02) plt.tight_layout() plt.show()
知识点测验

在Seaborn中，哪个函数可以创建分面图（FacetGrid）？

sns.subplot()
sns.FacetGrid() 或 sns.relplot()/catplot()
sns.multiplot()
sns.gridplot()

解析：sns.FacetGrid()是底层分面图函数，而sns.relplot()和sns.catplot()是更高级的封装，可以一步完成分面图的创建。

六、配对关系图

多变量关系矩阵

# 配对关系图：多变量关系一览 import seaborn as sns import matplotlib.pyplot as plt # 使用内置数据集 iris = sns.load_dataset('iris') # 创建配对关系图 g = sns.pairplot(iris, hue='species', palette={'setosa': '#2563eb', 'versicolor': '#10b981', 'virginica': '#f59e0b'}, diag_kind='kde', height=2.5) g.fig.suptitle('鸢尾花特征配对关系图', fontsize=14, fontweight='bold', y=1.02) plt.tight_layout() plt.show() print("\n配对关系图说明：") print("- 对角线：各特征的分布（KDE密度图）") print("- 非对角线：两两特征的散点图") print("- 颜色区分不同类别")

小贴士：配对关系图是数据探索的神器，可以一眼看出哪些特征对分类最有帮助。变量数量建议控制在5-6个以内，否则图表会过于庞大。

七、真实商业案例：用户行为数据分析

案例背景

某电商平台需要分析用户行为数据，了解不同用户群体的消费特征，为精准营销提供数据支持。

# 用户行为数据可视化分析 import seaborn as sns import matplotlib.pyplot as plt import pandas as pd import numpy as np # 生成模拟用户数据 np.random.seed(42) n_users = 500 user_data = pd.DataFrame({ '用户ID': range(1, n_users + 1), '用户等级': np.random.choice(['普通', 'VIP', 'SVIP'], n_users, p=[0.6, 0.3, 0.1]), '年龄': np.random.randint(18, 60, n_users), '月均消费': np.random.lognormal(4, 0.6, n_users), '购买频次': np.random.poisson(5, n_users) + 1, '客单价': np.random.lognormal(4.5, 0.4, n_users), '活跃天数': np.random.randint(5, 30, n_users) }) # 创建综合分析图表 fig = plt.figure(figsize=(16, 12)) # 1. 用户等级消费分布 ax1 = fig.add_subplot(2, 3, 1) sns.boxplot(data=user_data, x='用户等级', y='月均消费', palette={'普通': '#94a3b8', 'VIP': '#3b82f6', 'SVIP': '#f59e0b'}) ax1.set_title('各等级用户消费分布', fontweight='bold') # 2. 年龄与消费关系 ax2 = fig.add_subplot(2, 3, 2) sns.scatterplot(data=user_data, x='年龄', y='月均消费', hue='用户等级', alpha=0.6, palette={'普通': '#94a3b8', 'VIP': '#3b82f6', 'SVIP': '#f59e0b'}) ax2.set_title('年龄与消费关系', fontweight='bold') # 3. 购买频次分布 ax3 = fig.add_subplot(2, 3, 3) sns.histplot(data=user_data, x='购买频次', bins=15, kde=True, color='#2563eb') ax3.set_title('购买频次分布', fontweight='bold') # 4. 相关性热力图 ax4 = fig.add_subplot(2, 3, 4) numeric_cols = ['年龄', '月均消费', '购买频次', '客单价', '活跃天数'] corr = user_data[numeric_cols].corr() sns.heatmap(corr, annot=True, fmt='.2f', cmap='RdBu_r', center=0, ax=ax4) ax4.set_title('指标相关性', fontweight='bold') # 5. 客单价与购买频次 ax5 = fig.add_subplot(2, 3, 5) sns.scatterplot(data=user_data, x='购买频次', y='客单价', size='月均消费', sizes=(20, 200), alpha=0.6, color='#2563eb') ax5.set_title('频次-客单价-消费关系', fontweight='bold') # 6. 各等级用户数量 ax6 = fig.add_subplot(2, 3, 6) user_counts = user_data['用户等级'].value_counts() colors = ['#94a3b8', '#3b82f6', '#f59e0b'] ax6.pie(user_counts, labels=user_counts.index, autopct='%1.1f%%', colors=colors, wedgeprops=dict(width=0.5), startangle=90) ax6.set_title('用户等级占比', fontweight='bold') plt.suptitle('用户行为数据综合分析报告', fontsize=16, fontweight='bold', y=1.02) plt.tight_layout() plt.savefig('user_behavior_analysis.png', dpi=150, bbox_inches='tight') plt.show() print("\n分析报告已生成！")

案例要点：综合运用多种图表类型，从不同角度分析用户行为。箱线图展示分布差异，散点图揭示变量关系，热力图呈现相关性，饼图显示结构占比。
