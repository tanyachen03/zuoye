一、Matplotlib简介与环境配置

什么是Matplotlib？

Matplotlib是Python中最基础、最广泛使用的数据可视化库，由John D. Hunter于2003年创建。它提供了类似MATLAB的绘图接口，能够生成出版质量级别的图表。

  • 优势：功能强大、高度可定制、与NumPy/Pandas无缝集成
  • 应用场景：数据探索、报告制作、学术论文图表
  • 生态系统：Seaborn、Pandas绘图功能都基于Matplotlib构建

环境配置与导入

# 安装Matplotlib（如未安装） # pip install matplotlib # 标准导入方式 import matplotlib.pyplot as plt import numpy as np # 在Jupyter Notebook中显示图表 # %matplotlib inline # 设置中文字体（解决中文乱码问题） plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS'] plt.rcParams['axes.unicode_minus'] = False # 设置默认图表大小和分辨率 plt.rcParams['figure.figsize'] = (10, 6) plt.rcParams['figure.dpi'] = 100 print("Matplotlib版本:", plt.__version__)

小贴士：使用plt.style.use()可以快速应用预设样式，如'seaborn-v0_8-whitegrid'、'ggplot'等，让你的图表瞬间变得专业美观。

二、Matplotlib图表结构

Figure与Axes的关系

理解Matplotlib的层级结构是掌握绑图的关键：

  • Figure（画布）：整个图形的最外层容器，可以包含多个子图
  • Axes（子图）：单个图表区域，包含坐标轴、数据、标签等
  • Axis（坐标轴）：X轴和Y轴，控制刻度、标签和范围

# 两种创建图表的方式 import matplotlib.pyplot as plt # 方式1：简单方式（适合快速绘图） plt.figure(figsize=(8, 5)) plt.plot([1, 2, 3, 4], [1, 4, 2, 3]) plt.title('简单折线图') plt.show() # 方式2：面向对象方式（推荐，更灵活） fig, ax = plt.subplots(figsize=(8, 5)) ax.plot([1, 2, 3, 4], [1, 4, 2, 3]) ax.set_title('面向对象折线图') plt.show() # 创建多子图 fig, axes = plt.subplots(2, 2, figsize=(10, 8)) # axes是一个2x2的数组，通过axes[0,0], axes[0,1]等访问

常见错误：混淆plt.title()和ax.set_title()。前者是MATLAB风格的全局函数，后者是面向对象方法。建议统一使用面向对象方式，代码更清晰。

三、柱状图绑制

基础柱状图

柱状图是最常用的比较类图表，适合展示分类数据的数值大小。

# 柱状图完整示例 import matplotlib.pyplot as plt import numpy as np # 准备数据 categories = ['电子产品', '服装', '食品', '家居', '美妆'] sales = [156, 128, 95, 82, 68] # 创建图表 fig, ax = plt.subplots(figsize=(10, 6)) # 设置商务配色 colors = ['#2563eb', '#3b82f6', '#60a5fa', '#93c5fd', '#bfdbfe'] # 绑制柱状图 bars = ax.bar(categories, sales, color=colors, width=0.6, edgecolor='white', linewidth=1.2) # 添加数据标签 for bar in bars: height = bar.get_height() ax.annotate(f'{height}万', xy=(bar.get_x() + bar.get_width() / 2, height), xytext=(0, 5), textcoords='offset points', ha='center', va='bottom', fontsize=11, fontweight='bold') # 设置标题和坐标轴 ax.set_title('各品类季度销售额对比', fontsize=16, fontweight='bold', pad=20) ax.set_ylabel('销售额（万元）', fontsize=12) ax.set_xlabel('商品品类', fontsize=12) # 美化图表 ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.yaxis.grid(True, linestyle='--', alpha=0.3) ax.set_axisbelow(True) # 设置Y轴范围 ax.set_ylim(0, max(sales) * 1.15) plt.tight_layout() plt.savefig('bar_chart.png', dpi=150, bbox_inches='tight') plt.show() print("柱状图已保存！")

分组柱状图

# 分组柱状图：多系列对比 import matplotlib.pyplot as plt import numpy as np # 准备数据 quarters = ['Q1', 'Q2', 'Q3', 'Q4'] online_sales = [120, 145, 168, 195] offline_sales = [85, 78, 92, 88] x = np.arange(len(quarters)) width = 0.35 fig, ax = plt.subplots(figsize=(10, 6)) # 绑制两组柱状图 bars1 = ax.bar(x - width/2, online_sales, width, label='线上销售', color='#2563eb') bars2 = ax.bar(x + width/2, offline_sales, width, label='线下销售', color='#10b981') # 设置坐标轴 ax.set_xlabel('季度', fontsize=12) ax.set_ylabel('销售额（万元）', fontsize=12) ax.set_title('线上线下销售对比分析', fontsize=16, fontweight='bold', pad=20) ax.set_xticks(x) ax.set_xticklabels(quarters) ax.legend(loc='upper left', frameon=False) # 美化 ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.yaxis.grid(True, linestyle='--', alpha=0.3) plt.tight_layout() plt.show()
知识点测验

在Matplotlib中，哪个函数用于绑制柱状图？

plt.line()
plt.bar() 或 ax.bar()
plt.histogram()
plt.column()

解析：plt.bar()或ax.bar()用于绑制垂直柱状图，plt.barh()用于绑制水平条形图。

四、折线图绑制

基础折线图

折线图适合展示数据随时间的变化趋势，是时间序列分析的核心图表。

# 折线图完整示例 import matplotlib.pyplot as plt import numpy as np # 准备数据 months = ['1月', '2月', '3月', '4月', '5月', '6月'] sales_2023 = [120, 135, 148, 142, 158, 172] sales_2024 = [145, 162, 178, 185, 198, 215] # 创建图表 fig, ax = plt.subplots(figsize=(10, 6)) # 绑制折线图 ax.plot(months, sales_2023, marker='o', color='#94a3b8', linewidth=2, markersize=6, label='2023年') ax.plot(months, sales_2024, marker='o', color='#2563eb', linewidth=3, markersize=8, label='2024年') # 添加关键数据标注 ax.annotate(f'{sales_2024[-1]}万', xy=(months[-1], sales_2024[-1]), xytext=(10, 0), textcoords='offset points', fontsize=12, fontweight='bold', color='#2563eb') # 设置标题和坐标轴 ax.set_title('月度销售趋势对比', fontsize=16, fontweight='bold', pad=20) ax.set_ylabel('销售额（万元）', fontsize=12) ax.set_xlabel('月份', fontsize=12) ax.legend(loc='upper left', frameon=False) # 美化 ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.yaxis.grid(True, linestyle='--', alpha=0.3) # 计算并显示增长率 growth_rate = (sales_2024[-1] - sales_2023[-1]) / sales_2023[-1] * 100 ax.text(0.02, 0.98, f'同比增长: {growth_rate:.1f}%', transform=ax.transAxes, fontsize=11, verticalalignment='top', bbox=dict(boxstyle='round', facecolor='#f0fdf4', edgecolor='#10b981')) plt.tight_layout() plt.show()

面积图

# 面积图：强调累积效果 import matplotlib.pyplot as plt # 准备数据 months = ['1月', '2月', '3月', '4月', '5月', '6月'] revenue = [100, 120, 135, 142, 155, 168] costs = [65, 75, 82, 88, 95, 102] fig, ax = plt.subplots(figsize=(10, 6)) # 绑制填充区域 ax.fill_between(months, revenue, alpha=0.3, color='#2563eb', label='收入') ax.fill_between(months, costs, alpha=0.3, color='#ef4444', label='成本') # 绑制折线 ax.plot(months, revenue, color='#2563eb', linewidth=2, marker='o') ax.plot(months, costs, color='#ef4444', linewidth=2, marker='o') ax.set_title('收入与成本趋势分析', fontsize=16, fontweight='bold') ax.set_ylabel('金额（万元）', fontsize=12) ax.legend(loc='upper left', frameon=False) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) plt.tight_layout() plt.show()

小贴士：折线图中使用marker参数添加数据点标记，可以让读者更清楚地看到具体数据位置。常用marker样式：'o'圆点、's'方块、'^'三角形、'D'菱形。

五、饼图与环形图

基础饼图

# 饼图示例 import matplotlib.pyplot as plt # 准备数据 labels = ['电子产品', '服装', '食品', '家居', '其他'] sizes = [35, 25, 20, 12, 8] colors = ['#2563eb', '#3b82f6', '#60a5fa', '#93c5fd', '#bfdbfe'] explode = (0.05, 0, 0, 0, 0) # 突出第一块 fig, ax = plt.subplots(figsize=(8, 8)) wedges, texts, autotexts = ax.pie(sizes, explode=explode, labels=labels, colors=colors, autopct='%1.1f%%', startangle=90, pctdistance=0.75, shadow=False) # 设置字体 for text in texts: text.set_fontsize(12) for autotext in autotexts: autotext.set_fontsize(11) autotext.set_fontweight('bold') ax.set_title('产品销售占比分析', fontsize=16, fontweight='bold', pad=20) plt.tight_layout() plt.show()

环形图（更现代的选择）

# 环形图：现代商务风格 import matplotlib.pyplot as plt # 准备数据 labels = ['新客户', '老客户', '会员客户'] sizes = [35, 45, 20] colors = ['#2563eb', '#10b981', '#f59e0b'] fig, ax = plt.subplots(figsize=(8, 8)) # 绑制环形图 wedges, texts, autotexts = ax.pie(sizes, labels=labels, colors=colors, autopct='%1.1f%%', startangle=90, pctdistance=0.75, wedgeprops=dict(width=0.5, edgecolor='white')) # 中心添加文字 ax.text(0, 0, '客户\n结构', ha='center', va='center', fontsize=16, fontweight='bold', color='#1e293b') ax.set_title('客户结构分析', fontsize=16, fontweight='bold') plt.tight_layout() plt.show()

常见错误：饼图扇区过多（超过6个）会导致难以阅读。建议将小比例项合并为"其他"类别，或者改用柱状图展示。

六、散点图与气泡图

散点图：展示变量关系

# 散点图：分析广告投入与销售额的关系 import matplotlib.pyplot as plt import numpy as np # 准备数据 np.random.seed(42) ad_spend = np.random.uniform(10, 100, 50) sales = ad_spend * 2.5 + np.random.normal(0, 20, 50) fig, ax = plt.subplots(figsize=(10, 6)) # 绑制散点图 scatter = ax.scatter(ad_spend, sales, c='#2563eb', s=60, alpha=0.6, edgecolors='white', linewidths=1) # 添加趋势线 z = np.polyfit(ad_spend, sales, 1) p = np.poly1d(z) ax.plot(ad_spend, p(ad_spend), color='#ef4444', linewidth=2, linestyle='--', label='趋势线') ax.set_title('广告投入与销售额关系分析', fontsize=16, fontweight='bold') ax.set_xlabel('广告投入（万元）', fontsize=12) ax.set_ylabel('销售额（万元）', fontsize=12) ax.legend(frameon=False) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) # 计算相关系数 correlation = np.corrcoef(ad_spend, sales)[0, 1] ax.text(0.02, 0.98, f'相关系数: {correlation:.2f}', transform=ax.transAxes, fontsize=11, verticalalignment='top') plt.tight_layout() plt.show()

气泡图：三维数据展示

# 气泡图：展示三维数据 import matplotlib.pyplot as plt import numpy as np # 准备数据 products = ['产品A', '产品B', '产品C', '产品D', '产品E'] price = [50, 80, 120, 200, 150] # 价格 volume = [500, 350, 200, 100, 280] # 销量 profit_margin = [15, 25, 35, 45, 30] # 利润率（决定气泡大小） fig, ax = plt.subplots(figsize=(10, 6)) # 绑制气泡图 scatter = ax.scatter(price, volume, s=[p*10 for p in profit_margin], # 气泡大小 c=profit_margin, # 颜色映射 cmap='Blues', alpha=0.6, edgecolors='white', linewidths=2) # 添加产品标签 for i, product in enumerate(products): ax.annotate(product, (price[i], volume[i]), textcoords='offset points', xytext=(0, 10), ha='center') # 添加颜色条 cbar = plt.colorbar(scatter) cbar.set_label('利润率(%)', fontsize=11) ax.set_title('产品价格-销量-利润分析', fontsize=16, fontweight='bold') ax.set_xlabel('价格（元）', fontsize=12) ax.set_ylabel('销量（件）', fontsize=12) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) plt.tight_layout() plt.show()
